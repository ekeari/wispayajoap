  <script>
    const tg = window.Telegram?.WebApp || {};
    if (tg.expand) tg.expand();

    let userPhone = "";
    let userPin = "";
    let activeTier = "2k";
    let activeGroup = "Group A";
    let selectedSlots = [];

    const mockUserDatabase = {
      "08012345678": "1234"
    };

    function authenticateUser() {
      const phone = document.getElementById('phone').value.trim();
      const pin = document.getElementById('pin').value.trim();

      if (phone.length < 10 || pin.length !== 4) {
        showAuthError("Provide a valid WhatsApp Number and 4-Digit PIN.");
        return;
      }

      if (mockUserDatabase[phone] && mockUserDatabase[phone] !== pin) {
        showAuthError("Incorrect PIN! To reset/change your PIN, please contact <a href='https://t.me/wispay_support' style='color: var(--accent);'>Customer Service</a>.");
        return;
      }

      mockUserDatabase[phone] = pin;
      userPhone = phone;
      userPin = pin;

      document.getElementById('auth-screen').classList.add('hidden');
      document.getElementById('dashboard-screen').classList.remove('hidden');
      renderGrid();
    }

    function showAuthError(msg) {
      const errDiv = document.getElementById('auth-error');
      errDiv.innerHTML = msg;
      errDiv.classList.remove('hidden');
    }

    function setTier(tier, element) {
      activeTier = tier;
      document.querySelectorAll('.tabs .tab').forEach(t => t.classList.remove('active'));
      element.classList.add('active');
      renderGrid();
    }

    function setGroup(group) {
      activeGroup = group;
      document.getElementById('group-a-btn').classList.toggle('active', group === 'Group A');
      document.getElementById('group-b-btn').classList.toggle('active', group === 'Group B');
      renderGrid();
    }

    function renderGrid() {
      const grid = document.getElementById('slots-grid');
      grid.innerHTML = '';
      const totalSlots = activeTier === '50k' ? 5 : 10;
      
      for (let i = 1; i <= totalSlots; i++) {
        const slotKey = `${activeTier}-${activeGroup}-Slot#${i}`;
        const slotDiv = document.createElement('div');
        const isSelected = selectedSlots.includes(slotKey);
        
        slotDiv.className = `slot ${isSelected ? 'selected' : ''}`;
        slotDiv.innerText = `#${i}`;
        slotDiv.onclick = () => toggleSlot(slotKey);
        grid.appendChild(slotDiv);
      }
    }

    function toggleSlot(slotKey) {
      const index = selectedSlots.indexOf(slotKey);
      if (index > -1) {
        selectedSlots.splice(index, 1);
      } else {
        if (selectedSlots.length >= 3) {
          alert("Maximum Limit Reached: You can only select up to 3 slots total.");
          return;
        }
        selectedSlots.push(slotKey);
      }
      document.getElementById('slot-count').innerText = selectedSlots.length;
      renderGrid();
      updateScheduleCalendar();
    }

    function updateScheduleCalendar() {
      const list = document.getElementById('schedule-list');
      if (selectedSlots.length === 0) {
        list.innerHTML = '<p style="font-size: 12px;">No slots reserved yet. Select positions above to generate your payout calendar.</p>';
        return;
      }

      let html = '';
      selectedSlots.forEach(slotKey => {
        const [tier, group, slotNum] = slotKey.split('-');
        const posInt = parseInt(slotNum.replace('Slot#', ''));
        
        let splitInfo = "100% Full Payout at 2:00 PM";
        if (tier === "50k" && posInt <= 3) {
          splitInfo = "70% Initial Payout at 2:00 PM (30% Balance paid at cycle end)";
        } else if (tier !== "50k" && posInt <= 5) {
          splitInfo = "70% Initial Payout at 2:00 PM (30% Balance paid at cycle end)";
        }

        html += `
          <div class="schedule-item">
            <strong>${tier.toUpperCase()} ${group} — ${slotNum}</strong><br/>
            <span>Contribution Due: Before 12:00 PM (Noon)</span><br/>
            <span style="color: var(--accent);">Payout Status: ${splitInfo}</span>
          </div>
        `;
      });
      list.innerHTML = html;
    }

    function uploadProof() {
      const fileInput = document.getElementById('proof-file');
      if (!fileInput.files.length) {
        alert("Please select a proof image file before submitting.");
        return;
      }
      
      document.getElementById('proof-status').innerHTML = '<span class="badge badge-warn">⌛ Verification Submitted — Awaiting Admin Review</span>';
      
      if (tg.sendData) {
        tg.sendData(JSON.stringify({
          type: "BANK_PROOF_SUBMITTED",
          phone: userPhone,
          timestamp: new Date().toISOString()
        }));
      }
    }

    function confirmSlots() {
      if (selectedSlots.length === 0) {
        alert("Select at least 1 position slot before submitting.");
        return;
      }

      if (tg.sendData) {
        tg.sendData(JSON.stringify({
          type: "SLOTS_RESERVED",
          phone: userPhone,
          slots: selectedSlots
        }));
      }
      
      alert("Slots submitted successfully! Pending Admin verification.");
    }

    function submitPayoutAccount() {
      const bank = document.getElementById('payout-bank').value.trim();
      const accNum = document.getElementById('payout-account-num').value.trim();
      const accName = document.getElementById('payout-account-name').value.trim();

      if (!bank || accNum.length !== 10 || !accName) {
        alert("Please fill in a valid Bank Name, 10-digit Account Number, and Account Holder Name.");
        return;
      }

      document.getElementById('payout-account-status').innerHTML = 
        '<span class="badge badge-warn">✅ Payout Details Submitted! Expect transfer before 2:00 PM.</span>';

      if (tg.sendData) {
        tg.sendData(JSON.stringify({
          type: "PAYOUT_ACCOUNT_SUBMITTED",
          phone: userPhone,
          bankName: bank,
          accountNumber: accNum,
          accountName: accName,
          timestamp: new Date().toISOString()
        }));
      }
    }
